and Ellipse = {

    let id = nextContentsID()
    let s0 = Style ([{Key="visibility";Value="hidden"}]@s.list)

        html.taga ("ellipse", [Atr("id",id);]@[s0.atr])
    member this.ID with get() = id

        let t = Var_e "t"

 \begin{lstlisting}[caption={円弧を描画するアニメーション},label={Arc},breaklines=true,breakatwhitespace=false]
and Arc = {
    /// 円弧の中心座標
    center:tposition;
    /// 開始角（度数法, 反時計回りに描画）
    angle1:num0->num0;
    /// 終了角（度数法, 反時計回りに描画）
    angle2:num0->num0;
    /// 円弧の半径
    radius:num0->num0;}
and AnimationArc(s:Style,canvasX:int,canvasY:int) =
    let id = nextContentsID()
    let s0 = Style ([{Key="visibility";Value="hidden"}]@s.list)
    let s1 = Style ([{Key="visibility";Value="visible"}]@s.list)
    do
        html.taga ("path", [Atr("id",id);]@[s0.atr])
    member this.ID with get() = id
    member this.P (e:Arc) = 
        let t = Var_e "t"
        wrJS.Write ("    var e = document.getElementById(\""+id+"\");")
        let a1 = Math.PI * e.angle1 t / 180
        let x1 = e.center.X t + e.radius t * asm.cos a1
        let y1 = e.center.Y t + e.radius t * asm.sin a1
        wrJS.Write ("    var x1 = " + x1.str JavaScript+";")
        wrJS.Write ("    var y1 = " + (canvasY - y1).str JavaScript+";")
        let a2 = Math.PI * e.angle2 t / 180 - 1E-4
        let x2 = e.center.X t + e.radius t * asm.cos a2
        let y2 = e.center.Y t + e.radius t * asm.sin a2
        wrJS.Write ("    var x2 = " + x2.str JavaScript+";")
        wrJS.Write ("    var y2 = " + (canvasY - y2).str JavaScript+";")
        wrJS.Write ("    var a1 = " + (e.angle1 t).str JavaScript+";")
        wrJS.Write ("    var a2 = " + (e.angle2 t).str JavaScript+";")
        wrJS.Write ("    var radiusX = " + (e.radius t).str JavaScript + ";")
        wrJS.Write ("    var radiusY = " + (e.radius t).str JavaScript + ";")
        wrJS.Write  "    var da = a2 - a1;"
        wrJS.Write  "    if(da < 0.0) {da = a2 + 360 - a1;}"
        wrJS.Write  "    var largerOrSmaller = 0;"
        wrJS.Write  "    if(da > 180.0) {largerOrSmaller = 1;}"
        wrJS.Write ("    d = \"M \" + x1 + \" \" + y1 + \" A \" + radiusX + \" \" + radiusY + \" 0 \" + largerOrSmaller + \" 0 \" + x2 + \" \" + y2 " + ";")
        wrJS.Write ("    e.setAttribute(\"style\"," + "\"" + s1.code0 + "\");")
        wrJS.Write ("    e.setAttribute(\"d\", " + "d" + ");")
        
        wrJSAnimationSeqReset.Write ("    var e = document.getElementById(\""+id+"\");")
        wrJSAnimationSeqReset.Write ("    e.setAttribute(\"style\"," + "\"" + s0.code0 + "\");")
